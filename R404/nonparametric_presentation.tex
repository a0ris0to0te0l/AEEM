		What actually VaR is, and how it is similar?
		\begin{itemize}
			\item Value at Risk (VaR) is a financial metric used to assess the potential loss in value of an asset or portfolio over a defined period for a given confidence interval. The concept is typically applied in risk management to quantify the level of financial risk within a firm or investment portfolio. 
			\item One tailed test, usually set at 0.05 or 0.01 percentile.
		\end{itemize}
		\begin{theorem}[Value at Risk]
			$ \text{VaR}_{\alpha} = -\inf\{ x \in \mathbb{R} : P(X \leq x) \geq \alpha \}$
		\end{theorem}

	\subsection{Theory}
	\begin{frame}
		\frametitle{Kernel Density Estimation (KDE)}
		\framesubtitle{Theory}
		Kernel Density Estimation (KDE) is a non-parametric way to estimate the probability density function of a random variable. It smooths the observed data points to create a continuous probability density function.
		
		$$  \hat{f}(u) = \frac{1}{nh} \sum\nolimits_{i=1}^{n} K \left( \frac{u - u_i}{h} \right) $$ 
	\end{frame}

	\begin{frame}
		\frametitle{Bandwidth Selection}
		\framesubtitle{Theory}
		Three most used bandwidths are:
		\begin{itemize}
			\item Normal Reference Distribution (nrd / 0) $$ h_{nrd} = 0.9 sn^{-\frac{1}{5}}, \text{when s= 0, then h=1 in nrd0}$$
			\item Sheather - Jones (SJ) $$ h_{SJ} = s \left( \frac{4}{3n} \right)^{\frac{1}{5}} $$
			\item Unbiased Cross Validation (ucv) $$ h_{ucv} = arg \min_h \left( \frac{1}{n} \sum\nolimits_{i=1}^{n} K_h (x_i - u) f(u) du - \frac{1}{n} \sum\nolimits_{i=1}^{n} K_h (x_i) \right) $$
		\end{itemize}
		
	\end{frame}
	
	
	\begin{frame}
		\frametitle{Kernel Selection}
		\framesubtitle{Theory}
		   \begin{itemize}
			\item Rectangular - $\mathbb{I}$ - flat influence over the bandwidth.
			$ K(u) = \frac{1}{2} $
			 			
			\item Epanechnikov - $\mathbb{I}$ - Best performing for MISE. \quad $ K(u) = \frac{3}{4}(1 - u^2)$
		
			\item Biweight - $\mathbb{I}$ - smoother estimate. \quad 
			$ K(u) = \frac{15}{16}(1 - u^2)^2 $
			
			\item Triangular - $\mathbb{I}$ - Balance between ease of computation and smoothness
			$$ K(u) = (1 - |u|) $$
			
			\item Gaussian 	$$ K(u) = \frac{1}{\sqrt{2\pi}\sigma'} e^{-\frac{u^2}{2}} $$
		\end{itemize}
	\end{frame}
	
	\begin{frame}
		\frametitle{Local Linear Regression}
		\framesubtitle{Theory}
		Local Linear Regression is a non-parametric regression method that fits a linear function to a local subset of the data around each point of interest. This approach allows for flexibility in estimating complex relationships in the data.
		$$
		\hat{y}(x_0) = \frac{\sum_{i=1}^{n} K\left(\frac{x_i - x_0}{h}\right) y_i}{\sum_{i=1}^{n} K\left(\frac{x_i - x_0}{h}\right)}
		$$
	\end{frame}
	
	\begin{frame}
		\frametitle{Quintile Regression}
		\framesubtitle{Theory}
		Quintile Regression is a statistical technique that estimates the conditional median or other quantiles of the response variable, allowing for a comprehensive analysis of the relationship between variables beyond just the mean.
		$$
		    Q_y(\tau | X) = \frac{\sum_{i=1}^{n} K\left(\frac{y_i - Q_y(\tau | X)}{h}\right) y_i}{\sum_{i=1}^{n} K\left(\frac{y_i - Q_y(\tau | X)}{h}\right)}
		$$
	\end{frame}
	
	\begin{frame}
		\frametitle{Mean Integrated Square Error}
		\framesubtitle{Theory}
		The Mean Integrated Square Error (MISE) is used to evaluate the performance of density estimators. It takes into account both bias and variance, measuring how well the estimated density approximates the true density.
		$$
		\text{MISE} = \int \left( \hat{f}(x) - f(x) \right)^2 dx
		$$
		$$
		\text{MISE} = \text{Bias}^2 + \text{Variance}
		$$
	\end{frame}
