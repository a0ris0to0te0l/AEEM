	\subsection{Defining the problematic IV}
	\begin{frame}[fragile]
		\frametitle{Defining the main concern in IV}
		\framesubtitle{Invalid and Weak Instruments}
		\textbf{Instrument Variable}
		estimation is used to avoid biased results that arise when independent variables are correlated with the error term. This issue is addressed using a two-stage estimation (2SLS) or multi stage estimation process, where the first stage predicts the problematic variable with instruments, and the second or the additional stages uses these predictions to estimate the outcome variable.
		\\
		\vspace{1em}
		\textbf{Invalid Instrument Variable}
		are IVs that are correlated with the error term.\\
		\vspace{1em}
		\textbf{Weak Instrument Variable}
		are those that are weakly correlated with the independent variable, and thus do not effectively correct for biases.
	\end{frame}

\section{Invalid IVs}
\begin{frame}[fragile]
	\frametitle{Invalid IVs}
	\framesubtitle{Ensuring validity of Instrumental Variables (IVs)}
	\textbf{Relevance}\\
	Instruments must be correlated with the endogenous explanatory variable (incarceration rates, police numbers.)
	
	\vspace{1em}
	\textbf{Exclusion Restriction}\\
	instruments should affect the dependent variable (e.g crime rate) only trough the endogenous variable.
	
	\vspace{1em}
	\textbf{Preclusion of Links to Disturbances}\\
	verify empirical checks (lawsuits predicting past crime rates) that instruments are not related to unobserved shocks affecting the outcome.
\end{frame}

\begin{frame}[fragile]
	\frametitle{Invalid IVs}
	\framesubtitle{Strategies to Validate Instruments and Avoid Bias}
	\textbf{Test over-identifying restrictions}\\
	Use tests like Sargan’s to check if instruments are consistent with the data.
	$$ S = n * R^2 $$
	
	\vspace{1em}
	\textbf{Use Multiple Instruments}\\
	Comparing results from different instruments (e.g., election cycles vs. firefighters) enhances credibility; consistent estimates support validity.
	
	\vspace{1em}
	\textbf{Diligent Omitted Variable Checks}\\
	Include relevant controls to prevent omitted variable bias, ensuring that instruments are not correlated with unobserved factors.
\end{frame}
\begin{frame}[fragile]
	\frametitle{Invalid IVs}
	\framesubtitle{Additional Best Practices}
	\textbf{Leverage Intuition and Reduced-Form Regressions}\\
	Confirm that instruments behave as expected (e.g., litigation increases prison populations, which then reduce crime).
	
	\vspace{1em}
	\textbf{Avoid Data Mining}\\
	Pretesting and selecting instruments based on theoretical reasoning rather than fishing for significance reduces invalidity risk.
	
	\vspace{1em}
	\textbf{Caution with Over-Identification Tests}\\
	Recognize that failing to reject the null does not guarantee validity, especially if instruments share common rationales.
\end{frame}


\section{Weak IVs}
\begin{frame}[fragile]
	\frametitle{Weak IVs}
	\framesubtitle{Problems caused by them}
	\textbf{Finite-sample bias}\\
	two-stage least squares (2SLS) estimates can be biased, sometimes more than ordinary least squares (OLS).
	
	\vspace{1em}
	\textbf{Underestimated standard errors}\\
	Leads to overstated statistical significance.
	
	\vspace{1em}
	\textbf{Distorted inference}\\
	Confidence intervals and hypothesis tests become unreliable, especially in over-identified models.
\end{frame}

\begin{frame}[fragile]
	\frametitle{Weak IVs}
	\framesubtitle{Strategies to Address Weak Instruments}
	\textbf{Stock-Yogo Test}\\
	Checks if instruments are strong enough to produce reliable estimates (e.g., bias less than 10\% of OLS).
	
	\vspace{1em}
	\textbf{Conditional likelihood ratio tests (Moreira, Andrews)}\\
	Adjust for weak instruments, providing valid hypothesis testing and confidence intervals.
	$$ LRT = -2 ln \left(\frac{L_0}{L_1}\right) $$
	
	\vspace{1em}
	\textbf{Avoid relying solely on 2SLS}\\
	Always evaluate instrument strength before inference; avoid relying solely on 2SLS when instruments are weak.
\end{frame}
\section{Conclusions}

\begin{frame}[fragile] 
	\frametitle{Conclusion} 
	\textbf{Key Takeaways} \\
	
	The validity and strength of instruments are crucial for credible IV estimates. Weak or invalids IVs can lead to significant biases and unreliable inferences, especially when instruments are "almost valid" but weak.\\
	\vspace{1em}	
	Rigorous testing and validation are essential to avoid misleading results. 
\end{frame}

\begin{frame}[fragile]
	 \frametitle{Final Remarks} 
	 \textbf{Navigating Heterogeneity and Instrumental Variable Challenges} \
	
	Addressing heterogeneity requires understanding which aspects of responses are economically relevant and choosing appropriate econometric techniques.\\
	\vspace{1em}
	Recognizing that IV estimates may not reflect average effects across populations underscores the importance of careful instrument selection and interpretation. Diligence, theoretical grounding, and robustness checks are vital for credible causal inference. 
\end{frame}
